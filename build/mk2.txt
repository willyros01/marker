#!/usr/bin/env bash
# The Marker — Firebase set-up, prepared by Claude, Oct 16, 2026
#
# Apply:    bash mk2.txt
# Verify:   bash mk2.txt verify
# Rollback: bash mk2.txt rollback
#
# Replaces mk1.txt, which must NOT be run.
#
# What it does (it shows the plan and waits for you to type yes):
#   1. checks the published Firestore rules are exactly The Scorecard's
#      current rules (saved for rollback);
#   2. publishes the same rules plus The Marker's four small additions:
#      phones may drop a finished card off (markerSubmissions) and read the
#      course library and the worker's address; nothing else. Phones still
#      cannot write a single Scorecard round for anybody else;
#   3. switches on anonymous sign-in (for playing without an account);
#   4. creates two service accounts, Firestore only:
#        marker-worker   - the Cloudflare worker posts cards as this one;
#        marker-deployer - the deploy job; it may also make and delete keys
#                          for marker-worker, and nothing else;
#   5. lets the marker repository's main branch sign in as marker-deployer
#      with no key stored anywhere (keyless sign-in, like The Scorecard's
#      deletion job);
#   6. checks a key can be made for marker-worker (makes one, deletes it).
# No data is changed. If publishing the rules fails, the earlier rules are put
# back by themselves. Rollback undoes every step.
#
# Run it in Google Cloud Shell, signed in with the Google account that owns
# the Firebase project scorecard-f41b8.

# ---------------------------------------------------------------------------
# THE GOOGLE PROJECT — defined first, before any command runs.
# ---------------------------------------------------------------------------
PROJECT_ID="scorecard-f41b8"
PROJECT_NUMBER="811267714235"
unset CLOUDSDK_CORE_PROJECT GOOGLE_CLOUD_PROJECT

set -Eeuo pipefail

echo "Switching Cloud Shell to project ${PROJECT_ID} ..."
gcloud config set project "${PROJECT_ID}" --quiet >/dev/null 2>&1 || { echo "ERROR: could not switch to project ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
CURRENT_PROJECT="$(gcloud config get-value project 2>/dev/null)"
[[ "${CURRENT_PROJECT}" == "${PROJECT_ID}" ]] || { echo "ERROR: Cloud Shell is on '${CURRENT_PROJECT}', not ${PROJECT_ID}. Nothing changed." >&2; exit 1; }
echo "OK: Cloud Shell is now on project ${CURRENT_PROJECT}."
export CLOUDSDK_CORE_PROJECT="${PROJECT_ID}"
export GOOGLE_CLOUD_PROJECT="${PROJECT_ID}"

MODE="${1:-apply}"
COMMIT="__PIN__"                                 # willyros01/marker, tested: the rules
RAW="https://raw.githubusercontent.com/willyros01/marker"
STATE="$(pwd)/marker-mk2-backup"
RULES_API="https://firebaserules.googleapis.com/v1"
IDT_API="https://identitytoolkit.googleapis.com/admin/v2/projects/${PROJECT_ID}/config"
WORKER_NAME="marker-worker"
WORKER="${WORKER_NAME}@${PROJECT_ID}.iam.gserviceaccount.com"
SA_NAME="marker-deployer"
SA="${SA_NAME}@${PROJECT_ID}.iam.gserviceaccount.com"
POOL="scorecard-github"
PROVIDER="marker-provider"
MEMBER="principalSet://iam.googleapis.com/projects/${PROJECT_NUMBER}/locations/global/workloadIdentityPools/${POOL}/attribute.repository/willyros01/marker"

die(){ echo "ERROR: $*" >&2; exit 1; }
for c in gcloud curl jq python3; do command -v "$c" >/dev/null || die "$c is unavailable. Run this in Google Cloud Shell."; done

ACCOUNT="$(gcloud auth list --filter=status:ACTIVE --format='value(account)' | head -n1)"
[[ -n "${ACCOUNT}" ]] || die "No active Google account. Sign in to Cloud Shell first."
FOUND_ID="$(gcloud projects describe "${PROJECT_ID}" --project="${PROJECT_ID}" --format='value(projectId)' 2>/dev/null || true)"
[[ "${FOUND_ID}" == "${PROJECT_ID}" ]] || die "Project ${PROJECT_ID} was not found, or ${ACCOUNT} cannot see it. Nothing changed."
echo
echo "  Google account : ${ACCOUNT}"
echo "  Project        : ${PROJECT_ID}"
echo

token(){ gcloud auth print-access-token; }
HDR_PROJECT="x-goog-user-project: ${PROJECT_ID}"
get(){ curl --fail-with-body --silent --show-error -H "Authorization: Bearer $(token)" -H "${HDR_PROJECT}" "$1"; }
send(){ curl --fail-with-body --silent --show-error -X "$1" -H "Authorization: Bearer $(token)" -H "${HDR_PROJECT}" -H 'Content-Type: application/json' --data-binary "@$3" "$2"; }
release_ruleset(){ get "${RULES_API}/projects/${PROJECT_ID}/releases/cloud.firestore" | jq -er .rulesetName; }
save_rules(){ get "${RULES_API}/$1" | jq -er '.source.files[]|select(.name=="firestore.rules")|.content' > "$2"; [[ -s "$2" ]] || die "Empty Firestore rules."; }
same_rules(){ python3 - "$1" "$2" <<'PYC'
import sys
a=open(sys.argv[1]).read().rstrip("\n"); b=open(sys.argv[2]).read().rstrip("\n")
sys.exit(0 if a==b else 1)
PYC
}
publish_file(){
  local body="${STATE}/ruleset.json" patch="${STATE}/release.json" ruleset
  jq -n --rawfile rules "$1" '{source:{files:[{name:"firestore.rules",content:$rules}]}}' > "${body}" || return 1
  # Creating the ruleset compiles it; a rules error stops here, before anything is published.
  ruleset="$(send POST "${RULES_API}/projects/${PROJECT_ID}/rulesets" "${body}" | jq -er .name)" || return 1
  [[ -n "${ruleset}" ]] || return 1
  jq -n --arg name "projects/${PROJECT_ID}/releases/cloud.firestore" --arg ruleset "${ruleset}" '{release:{name:$name,rulesetName:$ruleset},updateMask:"rulesetName"}' > "${patch}" || return 1
  send PATCH "${RULES_API}/projects/${PROJECT_ID}/releases/cloud.firestore" "${patch}" >/dev/null || return 1
}
live_is(){
  local current
  current="$(release_ruleset 2>/dev/null)" || return 1
  get "${RULES_API}/${current}" 2>/dev/null | jq -er '.source.files[]|select(.name=="firestore.rules")|.content' > "${STATE}/rules-live-now.txt" 2>/dev/null || return 1
  same_rules "$1" "${STATE}/rules-live-now.txt"
}
recover(){
  local what="$1" tries=0 back=no
  echo
  echo "Something failed during ${what}. Putting the earlier rules back automatically ..."
  while (( tries < 3 )); do
    tries=$((tries + 1))
    if live_is "${STATE}/rules-before.txt"; then back=yes; break; fi
    publish_file "${STATE}/rules-before.txt" || true
    sleep 3
  done
  if [[ "${back}" == yes ]]; then
    echo "DONE: the earlier rules are live again. The Scorecard works exactly as before."
  else
    echo "WARNING: the earlier rules could not be confirmed. Run:  bash mk2.txt rollback  and send a screenshot to Claude."
  fi
  echo "Send a screenshot to Claude."
  exit 1
}
anon_on(){ get "${IDT_API}" | jq -r '.signIn.anonymous.enabled // false'; }
set_anon(){
  jq -n --argjson on "$1" '{signIn:{anonymous:{enabled:$on}}}' > "${STATE}/anon.json"
  send PATCH "${IDT_API}?updateMask=signIn.anonymous.enabled" "${STATE}/anon.json" >/dev/null
}
sa_exists(){ gcloud iam service-accounts describe "${1:-${SA}}" --project="${PROJECT_ID}" >/dev/null 2>&1; }
provider_exists(){ gcloud iam workload-identity-pools providers describe "${PROVIDER}" --project="${PROJECT_ID}" --location=global --workload-identity-pool="${POOL}" >/dev/null 2>&1; }
binding_exists(){ gcloud iam service-accounts get-iam-policy "${SA}" --project="${PROJECT_ID}" --format=json 2>/dev/null | jq -e --arg m "${MEMBER}" '.bindings[]? | select(.role=="roles/iam.workloadIdentityUser") | .members[] | select(.==$m)' >/dev/null; }
role_exists(){ gcloud projects get-iam-policy "${PROJECT_ID}" --format=json | jq -e --arg m "serviceAccount:${1:-${SA}}" '.bindings[] | select(.role=="roles/datastore.user") | .members[] | select(.==$m)' >/dev/null; }
keyadmin_exists(){ gcloud iam service-accounts get-iam-policy "${WORKER}" --project="${PROJECT_ID}" --format=json 2>/dev/null | jq -e --arg m "serviceAccount:${SA}" '.bindings[]? | select(.role=="roles/iam.serviceAccountKeyAdmin") | .members[] | select(.==$m)' >/dev/null; }
make_sa(){   # name, display name, marker file
  if sa_exists "$1@${PROJECT_ID}.iam.gserviceaccount.com"; then echo "OK: service account $1 already exists."
  else
    gcloud iam service-accounts create "$1" --project="${PROJECT_ID}" --display-name="$2" --quiet >/dev/null
    touch "${STATE}/$3"; sleep 5
    echo "OK: service account $1 created."
  fi
  if role_exists "$1@${PROJECT_ID}.iam.gserviceaccount.com"; then echo "OK: $1 already has Firestore access."
  else
    gcloud projects add-iam-policy-binding "${PROJECT_ID}" --member="serviceAccount:$1@${PROJECT_ID}.iam.gserviceaccount.com" --role="roles/datastore.user" --condition=None --quiet >/dev/null
    touch "${STATE}/$3-role"
    echo "OK: $1 has Firestore access (roles/datastore.user, nothing else)."
  fi
}
key_check(){   # makes one test key for marker-worker and deletes that same key (the worker's own key is left alone)
  local tmp id; tmp="$(mktemp)"
  gcloud iam service-accounts keys create "${tmp}" --iam-account="${WORKER}" --project="${PROJECT_ID}" --quiet >/dev/null 2>&1 || { rm -f "${tmp}"; return 1; }
  id="$(jq -r .private_key_id "${tmp}")"; rm -f "${tmp}"
  [[ -n "${id}" && "${id}" != null ]] || return 1
  gcloud iam service-accounts keys delete "${id}" --iam-account="${WORKER}" --project="${PROJECT_ID}" --quiet >/dev/null 2>&1 \
    || echo "NOTE: the test key ${id} could not be deleted; the next deploy deletes it."
  return 0
}

mkdir -p "${STATE}"
[[ "${COMMIT}" != "__PIN__" ]] || die "This copy of mk2.txt is not finished (no commit pinned). Ask Claude for the final one."
curl --fail --silent --show-error "${RAW}/${COMMIT}/firebase/firestore.rules" -o "${STATE}/rules-new.txt" || die "Cannot read the new rules from GitHub. Nothing changed."
curl --fail --silent --show-error "${RAW}/${COMMIT}/firebase/scorecard-live.rules" -o "${STATE}/rules-expected.txt" || die "Cannot read The Scorecard's rules from GitHub. Nothing changed."
grep -Fq "match /markerSubmissions/{cardId}" "${STATE}/rules-new.txt" || die "The downloaded rules are not The Marker's. Nothing changed."

if [[ "${MODE}" == apply ]]; then
  echo "THE PLAN"
  echo "  1. Firestore rules: The Scorecard's rules plus The Marker's four small additions."
  echo "  2. Anonymous sign-in: switched on (playing without an account)."
  echo "  3. Service accounts ${WORKER_NAME} and ${SA_NAME}: Firestore only;"
  echo "     ${SA_NAME} may also make and delete keys for ${WORKER_NAME}."
  echo "  4. Keyless sign-in for willyros01/marker (main branch only), as ${SA_NAME}."
  echo "  5. A check that a key can be made for ${WORKER_NAME} (made and deleted)."
  echo "  No data is changed."
  echo
  read -r -p "Type yes to go ahead: " ANSWER
  [[ "${ANSWER}" == "yes" ]] || die "Stopped. Nothing was changed."
  echo

  # 1. Rules.
  if live_is "${STATE}/rules-new.txt"; then
    echo "OK 1: The Marker's rules are already live."
  else
    save_rules "$(release_ruleset)" "${STATE}/rules-before.txt"
    same_rules "${STATE}/rules-before.txt" "${STATE}/rules-expected.txt" || die "The published rules are not The Scorecard's current rules, so nothing was changed. Send a screenshot to Claude."
    echo "OK: the published rules are The Scorecard's current rules (saved for rollback)."
    publish_file "${STATE}/rules-new.txt" || recover "publishing the rules"
    sleep 3
    live_is "${STATE}/rules-new.txt" || { sleep 10; live_is "${STATE}/rules-new.txt"; } || recover "checking the published rules"
    echo "OK 1: The Marker's rules are published."
  fi

  # 2. Anonymous sign-in.
  BEFORE="$(anon_on)" || die "Could not read the sign-in settings. The rules are published; run bash mk2.txt again."
  [[ -f "${STATE}/anon-before.txt" ]] || echo "${BEFORE}" > "${STATE}/anon-before.txt"
  if [[ "${BEFORE}" == "true" ]]; then echo "OK 2: anonymous sign-in was already on."
  else
    set_anon true || die "Could not switch on anonymous sign-in. Run bash mk2.txt again."
    [[ "$(anon_on)" == "true" ]] || die "Anonymous sign-in did not switch on. Run bash mk2.txt again."
    echo "OK 2: anonymous sign-in is on."
  fi

  # 3. Service accounts, Firestore only.
  make_sa "${WORKER_NAME}" "The Marker worker (Cloudflare)" worker-created
  make_sa "${SA_NAME}" "The Marker deploy job" deployer-created
  if keyadmin_exists; then echo "OK 3: ${SA_NAME} may already manage ${WORKER_NAME}'s keys."
  else
    gcloud iam service-accounts add-iam-policy-binding "${WORKER}" --project="${PROJECT_ID}" --role="roles/iam.serviceAccountKeyAdmin" --member="serviceAccount:${SA}" --quiet >/dev/null
    touch "${STATE}/keyadmin-added"
    echo "OK 3: ${SA_NAME} may make and delete keys for ${WORKER_NAME} (that account only)."
  fi

  # 4. Keyless sign-in for the marker repository's main branch.
  gcloud iam workload-identity-pools describe "${POOL}" --project="${PROJECT_ID}" --location=global >/dev/null 2>&1 \
    || die "The keyless sign-in pool ${POOL} was not found. Steps 1-3 are done; send a screenshot to Claude."
  if provider_exists; then echo "OK 4: keyless sign-in for willyros01/marker already exists."
  else
    gcloud iam workload-identity-pools providers create-oidc "${PROVIDER}" --project="${PROJECT_ID}" --location=global \
      --workload-identity-pool="${POOL}" --display-name="willyros01/marker main" \
      --issuer-uri="https://token.actions.githubusercontent.com" \
      --attribute-mapping="google.subject=assertion.sub,attribute.repository=assertion.repository,attribute.ref=assertion.ref" \
      --attribute-condition="assertion.repository=='willyros01/marker' && assertion.ref=='refs/heads/main'" --quiet >/dev/null
    touch "${STATE}/provider-created"
    echo "OK 4: keyless sign-in created, for willyros01/marker main only."
  fi
  if binding_exists; then echo "OK 4: the marker repository may already use ${SA_NAME}."
  else
    gcloud iam service-accounts add-iam-policy-binding "${SA}" --project="${PROJECT_ID}" --role="roles/iam.workloadIdentityUser" --member="${MEMBER}" --quiet >/dev/null
    touch "${STATE}/binding-added"
    echo "OK 4: the marker repository may use ${SA_NAME}."
  fi

  # 5. Keys can be made (an organisation policy can forbid it).
  if key_check; then echo "OK 5: a key for ${WORKER_NAME} can be made (the test key is deleted again)."
  else die "Google would not make a key for ${WORKER_NAME} (an organisation policy may forbid it). Steps 1-4 are done; send a screenshot to Claude."
  fi
  echo
  echo "DONE. All five steps are in place. Tell Claude."
elif [[ "${MODE}" == verify ]]; then
  ok=yes
  live_is "${STATE}/rules-new.txt" && echo "OK: The Marker's rules are live." || { echo "NOT YET: The Marker's rules are not live."; ok=no; }
  [[ "$(anon_on)" == "true" ]] && echo "OK: anonymous sign-in is on." || { echo "NOT YET: anonymous sign-in is off."; ok=no; }
  for n in "${WORKER_NAME}" "${SA_NAME}"; do
    sa_exists "$n@${PROJECT_ID}.iam.gserviceaccount.com" && echo "OK: service account $n exists." || { echo "NOT YET: no service account $n."; ok=no; }
    role_exists "$n@${PROJECT_ID}.iam.gserviceaccount.com" && echo "OK: $n has Firestore access." || { echo "NOT YET: $n has no Firestore access."; ok=no; }
  done
  keyadmin_exists && echo "OK: ${SA_NAME} may manage ${WORKER_NAME}'s keys." || { echo "NOT YET: ${SA_NAME} may not manage ${WORKER_NAME}'s keys."; ok=no; }
  provider_exists && echo "OK: keyless sign-in for willyros01/marker exists." || { echo "NOT YET: no keyless sign-in."; ok=no; }
  binding_exists && echo "OK: the marker repository may use it." || { echo "NOT YET: the marker repository may not use it."; ok=no; }
  [[ "${ok}" == yes ]] && echo "ALL OK." || exit 1
elif [[ "${MODE}" == rollback ]]; then
  read -r -p "Type yes to undo The Marker's set-up: " ANSWER
  [[ "${ANSWER}" == "yes" ]] || die "Stopped. Nothing was changed."
  if [[ -f "${STATE}/binding-added" ]]; then gcloud iam service-accounts remove-iam-policy-binding "${SA}" --project="${PROJECT_ID}" --role="roles/iam.workloadIdentityUser" --member="${MEMBER}" --quiet >/dev/null && rm -f "${STATE}/binding-added" && echo "OK: repository access removed."; fi
  if [[ -f "${STATE}/provider-created" ]]; then gcloud iam workload-identity-pools providers delete "${PROVIDER}" --project="${PROJECT_ID}" --location=global --workload-identity-pool="${POOL}" --quiet >/dev/null && rm -f "${STATE}/provider-created" && echo "OK: keyless sign-in removed."; fi
  if [[ -f "${STATE}/keyadmin-added" ]]; then gcloud iam service-accounts remove-iam-policy-binding "${WORKER}" --project="${PROJECT_ID}" --role="roles/iam.serviceAccountKeyAdmin" --member="serviceAccount:${SA}" --quiet >/dev/null && rm -f "${STATE}/keyadmin-added" && echo "OK: key access removed."; fi
  for pair in "${SA_NAME}:deployer-created" "${WORKER_NAME}:worker-created"; do
    n="${pair%%:*}"; f="${pair##*:}"; a="$n@${PROJECT_ID}.iam.gserviceaccount.com"
    if [[ -f "${STATE}/$f-role" ]]; then gcloud projects remove-iam-policy-binding "${PROJECT_ID}" --member="serviceAccount:$a" --role="roles/datastore.user" --condition=None --quiet >/dev/null && rm -f "${STATE}/$f-role" && echo "OK: $n's Firestore access removed."; fi
    if [[ -f "${STATE}/$f" ]]; then gcloud iam service-accounts delete "$a" --project="${PROJECT_ID}" --quiet >/dev/null && rm -f "${STATE}/$f" && echo "OK: service account $n removed (its keys go with it)."; fi
  done
  if [[ -f "${STATE}/anon-before.txt" && "$(cat "${STATE}/anon-before.txt")" != "true" ]]; then set_anon false && echo "OK: anonymous sign-in is off again."; fi
  if [[ -s "${STATE}/rules-before.txt" ]]; then
    publish_file "${STATE}/rules-before.txt" || die "The earlier rules could not be published. Send a screenshot to Claude."
    sleep 3
    live_is "${STATE}/rules-before.txt" || { sleep 10; live_is "${STATE}/rules-before.txt"; } || die "Rollback of the rules did not take. Send a screenshot to Claude."
    echo "OK: The Scorecard's earlier rules are published again."
  fi
  echo "DONE. Tell Claude."
else
  die "Unknown mode ${MODE}. Use: bash mk2.txt   or   bash mk2.txt verify   or   bash mk2.txt rollback"
fi
